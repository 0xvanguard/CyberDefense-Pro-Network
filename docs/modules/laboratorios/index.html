docker-compose.yml - DVWA + WebGoat
version: '3.8'
services:
  dvwa:
    image: vulnerables/web-dvwa
    container_name: dvwa
    ports: ["8080:80"]
    depends_on: [db]
    networks: [vuln-net]

  db:
    image: mysql:5.7
    environment:
      MYSQL_ROOT_PASSWORD: dvwa
      MYSQL_DATABASE: dvwa
    volumes: [db-data:/var/lib/mysql]
    networks: [vuln-net]

  webgoat:
    image: webgoat/webgoat
    ports: ["8081:8080", "9090:9090"]
    networks: [vuln-net]

networks:
  vuln-net: {driver: bridge}
volumes:
  db-data:
Comandos Esenciales
docker-compose up -d        # Levantar
docker ps                    # Ver contenedores
docker logs -f dvwa          # Ver logs
docker exec -it dvwa bash    # Entrar al contenedor
docker-compose down          # Detener
docker-compose down -v       # Limpiar todo

login.php - Vulnerable vs Seguro
<?php
// VULNERABLE (para practica)
$query = "SELECT * FROM users WHERE username='$user' AND pass='$pass'";

// SEGURO (produccion)
$stmt = $conn->prepare("SELECT * FROM users WHERE username=? AND pass=?");
$stmt->bind_param("ss", $user, $pass);
$stmt->execute();
?>
